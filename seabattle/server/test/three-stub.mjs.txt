export class Vector2 {
  constructor(x = 0, y = 0) { this.x = x; this.y = y }
  set(x, y) { this.x = x; this.y = y; return this }
  copy(v) { this.x = v.x; this.y = v.y; return this }
  clone() { return new Vector2(this.x, this.y) }
  add(v) { this.x += v.x; this.y += v.y; return this }
  sub(v) { this.x -= v.x; this.y -= v.y; return this }
  multiplyScalar(s) { this.x *= s; this.y *= s; return this }
  length() { return Math.hypot(this.x, this.y) }
  normalize() { const l = this.length() || 1; return this.multiplyScalar(1 / l) }
}
export class Vector3 {
  constructor(x = 0, y = 0, z = 0) { this.x = x; this.y = y; this.z = z }
  set(x, y, z) { this.x = x; this.y = y; this.z = z; return this }
  setScalar(s) { return this.set(s, s, s) }
  copy(v) { return this.set(v.x, v.y, v.z) }
  clone() { return new Vector3(this.x, this.y, this.z) }
  add(v) { this.x += v.x; this.y += v.y; this.z += v.z; return this }
  addScalar(s) { this.x += s; this.y += s; this.z += s; return this }
  addVectors(a, b) { return this.set(a.x + b.x, a.y + b.y, a.z + b.z) }
  addScaledVector(v, s) { this.x += v.x * s; this.y += v.y * s; this.z += v.z * s; return this }
  sub(v) { this.x -= v.x; this.y -= v.y; this.z -= v.z; return this }
  subVectors(a, b) { return this.set(a.x - b.x, a.y - b.y, a.z - b.z) }
  multiplyScalar(s) { this.x *= s; this.y *= s; this.z *= s; return this }
  divideScalar(s) { return this.multiplyScalar(1 / (s || 1)) }
  length() { return Math.hypot(this.x, this.y, this.z) }
  lengthSq() { return this.x ** 2 + this.y ** 2 + this.z ** 2 }
  normalize() { const l = this.length() || 1; return this.multiplyScalar(1 / l) }
  setLength(l) { return this.normalize().multiplyScalar(l) }
  distanceTo(v) { return Math.hypot(this.x - v.x, this.y - v.y, this.z - v.z) }
  dot(v) { return this.x * v.x + this.y * v.y + this.z * v.z }
  cross(v) { return this.crossVectors(this.clone(), v) }
  crossVectors(a, b) { return this.set(a.y * b.z - a.z * b.y, a.z * b.x - a.x * b.z, a.x * b.y - a.y * b.x) }
  lerp(v, t) { this.x += (v.x - this.x) * t; this.y += (v.y - this.y) * t; this.z += (v.z - this.z) * t; return this }
  lerpVectors(a, b, t) { return this.set(a.x + (b.x - a.x) * t, a.y + (b.y - a.y) * t, a.z + (b.z - a.z) * t) }
  applyQuaternion() { return this }
  applyAxisAngle() { return this }
  applyMatrix4() { return this }
  project() { return this }
  unproject() { return this }
  negate() { return this.multiplyScalar(-1) }
  toArray() { return [this.x, this.y, this.z] }
  equals(v) { return this.x === v.x && this.y === v.y && this.z === v.z }
}

let uid = 0
export class Object3D {
  constructor() {
    this.id = ++uid
    this.uuid = 'obj-' + this.id
    this.name = ''
    this.type = 'Object3D'
    this.children = []
    this.parent = null
    this.position = new Vector3()
    this.rotation = new Euler()
    this.scale = new Vector3(1, 1, 1)
    this.quaternion = new Quaternion()
    this.up = new Vector3(0, 1, 0)
    this.matrixWorld = new Matrix4()
    this.visible = true
    this.castShadow = false
    this.receiveShadow = false
    this.frustumCulled = true
    this.renderOrder = 0
    this.userData = {}
    this.layers = new Layers()
    this.isObject3D = true
  }
  add(...objs) { for (const o of objs) { if (!o) continue; o.parent = this; this.children.push(o) } return this }
  remove(...objs) { for (const o of objs) { const i = this.children.indexOf(o); if (i >= 0) { this.children.splice(i, 1); o.parent = null } } return this }
  clear() { this.children.length = 0; return this }
  traverse(cb) { cb(this); for (const c of this.children.slice()) c.traverse(cb) }
  traverseVisible(cb) { if (this.visible) this.traverse(cb) }
  getObjectByName(n) { let f = null; this.traverse(o => { if (!f && o.name === n) f = o }); return f }
  lookAt() { return this }
  updateMatrixWorld() { return this }
  updateMatrix() { return this }
  localToWorld(v) { return v }
  worldToLocal(v) { return v }
  getWorldPosition(v) { return (v || new Vector3()).copy(this.position) }
  attach(o) { return this.add(o) }
  removeFromParent() { if (this.parent) this.parent.remove(this); return this }
  raycast() {}
}

class BufferAttributeBase {
  constructor(array, itemSize) { this.array = array; this.itemSize = itemSize; this.count = array ? array.length / itemSize : 0; this.needsUpdate = false }
  setXYZ(i, x, y, z) { const a = this.array; if (a) { a[i * this.itemSize] = x; a[i * this.itemSize + 1] = y; a[i * this.itemSize + 2] = z } return this }
  setXY(i, x, y) { const a = this.array; if (a) { a[i * this.itemSize] = x; a[i * this.itemSize + 1] = y } return this }
  setXYZW(i, x, y, z, w) { this.setXYZ(i, x, y, z); if (this.array) this.array[i * this.itemSize + 3] = w; return this }
  setX(i, v) { if (this.array) this.array[i * this.itemSize] = v; return this }
  setY(i, v) { if (this.array) this.array[i * this.itemSize + 1] = v; return this }
  setZ(i, v) { if (this.array) this.array[i * this.itemSize + 2] = v; return this }
  setW(i, v) { if (this.array) this.array[i * this.itemSize + 3] = v; return this }
  getX(i) { return this.array ? this.array[i * this.itemSize] : 0 }
  getY(i) { return this.array ? this.array[i * this.itemSize + 1] : 0 }
  getZ(i) { return this.array ? this.array[i * this.itemSize + 2] : 0 }
  getW(i) { return this.array ? this.array[i * this.itemSize + 3] : 0 }
  setUsage() { return this }
  clone() { return new BufferAttributeBase(this.array ? this.array.slice() : null, this.itemSize) }
}

}

export const MathUtils = {
  clamp: (v, a, b) => Math.min(b, Math.max(a, v)),
  lerp: (a, b, t) => a + (b - a) * t,
  degToRad: d => d * Math.PI / 180,
  radToDeg: r => r * 180 / Math.PI,
  randFloat: (a, b) => a + Math.random() * (b - a),
  randFloatSpread: r => (Math.random() - 0.5) * r,
  smoothstep: (x, a, b) => { const t = Math.min(1, Math.max(0, (x - a) / (b - a))); return t * t * (3 - 2 * t) },
  euclideanModulo: (n, m) => ((n % m) + m) % m
}
